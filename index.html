prisma/schema.prisma (Production PostgreSQL Schema) 
// Prisma Schema for CelebriPlan Event Platform
datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

enum Role {
  ORGANIZER
  VENDOR
  GUEST
  ADMIN
}

enum RSVPStatus {
  ACCEPTED
  DECLINED
  PENDING
}

enum BookingStatus {
  REQUESTED
  QUOTED
  CONFIRMED
  CANCELLED
}

model User {
  id        String    @id @default(uuid())
  email     String    @unique
  name      String
  phone     String?
  role      Role      @default(ORGANIZER)
  events    Event[]   @relation("OrganizerEvents")
  vendor    Vendor?   @relation("VendorOwner")
  createdAt DateTime  @default(now())
}

model Event {
  id           String        @id @default(uuid())
  title        String
  type         String        // birthday, wedding, college-fest, corporate
  city         String
  budgetPaise  BigInt
  guestCount   Int
  theme        String
  durationHour Int           @default(4)
  organizerId  String
  organizer    User          @relation("OrganizerEvents", fields: [organizerId], references: [id])
  invitation   Invitation?
  tasks        Task[]
  expenses     Expense[]
  bookings     Booking[]
  createdAt    DateTime      @default(now())
}

model Invitation {
  id        String   @id @default(uuid())
  eventId   String   @unique
  event     Event    @relation(fields: [eventId], references: [id])
  qrCodeUrl String
  template  String   @default("elegant")
  guests    Guest[]
}

model Guest {
  id           String     @id @default(uuid())
  invitationId String
  invitation   Invitation @relation(fields: [invitationId], references: [id])
  name         String
  phone        String?
  email        String?
  rsvpStatus   RSVPStatus @default(PENDING)
  tableNumber  Int?
  checkedIn    Boolean    @default(false)
}

model Task {
  id          String   @id @default(uuid())
  eventId     String
  event       Event    @relation(fields: [eventId], references: [id])
  title       String
  priority    String   @default("medium")
  completed   Boolean  @default(false)
  dueDate     DateTime?
  gcalEventId String?
}

model Expense {
  id          String   @id @default(uuid())
  eventId     String
  event       Event    @relation(fields: [eventId], references: [id])
  category    String   // venue, food, decoration, photography, other
  description String
  amountPaise BigInt
  status      String   @default("paid")
}

model Vendor {
  id          String     @id @default(uuid())
  ownerId     String     @unique
  owner       User       @relation("VendorOwner", fields: [ownerId], references: [id])
  name        String
  category    String     // venue, decorator, photographer, caterer, dj
  city        String
  rating      Float      @default(5.0)
  services    Service[]
  bookings    Booking[]
}

model Service {
  id             String  @id @default(uuid())
  vendorId       String
  vendor         Vendor  @relation(fields: [vendorId], references: [id])
  name           String
  basePricePaise BigInt
  capacity       Int?
}

model Booking {
  id          String        @id @default(uuid())
  eventId     String
  event       Event         @relation(fields: [eventId], references: [id])
  vendorId    String
  vendor      Vendor        @relation(fields: [vendorId], references: [id])
  status      BookingStatus @default(CONFIRMED)
  totalPaise  BigInt
  advancePaid BigInt        @default(0)
  createdAt   DateTime      @default(now())
}
POST /api/ai/plan

Executes Section 9 AI Event Planning Engine. Accepts { type, city, guests, budgetPaise, theme, durationHours } and returns clean structured output with title, budget breakdown, tasks & timeline.

GET /api/vendors?category=&city=

Returns verified marketplace vendors filtered by category (venue, caterer, decorator, photographer, dj) and city.

POST /api/vendors/compare

Side-by-side comparative analysis of selected vendor IDs across capacity, pricing, ratings, and inclusions.

POST /api/guests/rsvp

Records guest RSVP status, generates unique QR entry pass code, and updates real-time attendance tracking.

GET /api/calendar/export

Generates compliant iCalendar (.ICS) file containing all event milestones and synchronized tasks for Google Calendar & Apple Calendar.
